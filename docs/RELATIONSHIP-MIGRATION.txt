-- Create persona_relationships table to track user-persona interaction depth
CREATE TABLE IF NOT EXISTS persona_relationships (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  persona_slug TEXT NOT NULL,
  conversation_count INTEGER DEFAULT 0,
  relationship_level INTEGER DEFAULT 1,
  last_conversation_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE(user_id, persona_slug)
);

-- Create index for faster lookups
CREATE INDEX IF NOT EXISTS idx_persona_relationships_user_persona 
ON persona_relationships(user_id, persona_slug);

-- Enable Row Level Security
ALTER TABLE persona_relationships ENABLE ROW LEVEL SECURITY;

-- RLS Policy: Users can only see their own relationships
CREATE POLICY "Users can view own relationships"
ON persona_relationships FOR SELECT
USING (auth.uid() = user_id);

-- RLS Policy: Users can insert their own relationships
CREATE POLICY "Users can insert own relationships"
ON persona_relationships FOR INSERT
WITH CHECK (auth.uid() = user_id);

-- RLS Policy: Users can update their own relationships
CREATE POLICY "Users can update own relationships"
ON persona_relationships FOR UPDATE
USING (auth.uid() = user_id);

-- Function to calculate relationship level based on conversation count
CREATE OR REPLACE FUNCTION calculate_relationship_level(conv_count INTEGER)
RETURNS INTEGER AS $$
BEGIN
  CASE
    WHEN conv_count >= 51 THEN RETURN 5;
    WHEN conv_count >= 31 THEN RETURN 4;
    WHEN conv_count >= 16 THEN RETURN 3;
    WHEN conv_count >= 6 THEN RETURN 2;
    ELSE RETURN 1;
  END CASE;
END;
$$ LANGUAGE plpgsql;

-- Function to update relationship after each conversation
CREATE OR REPLACE FUNCTION update_persona_relationship()
RETURNS TRIGGER AS $$
BEGIN
  -- Only process when a new conversation is created
  IF TG_OP = 'INSERT' AND NEW.user_id IS NOT NULL THEN
    INSERT INTO persona_relationships (user_id, persona_slug, conversation_count, relationship_level, last_conversation_at)
    VALUES (
      NEW.user_id,
      NEW.persona_slug,
      1,
      1,
      NOW()
    )
    ON CONFLICT (user_id, persona_slug)
    DO UPDATE SET
      conversation_count = persona_relationships.conversation_count + 1,
      relationship_level = calculate_relationship_level(persona_relationships.conversation_count + 1),
      last_conversation_at = NOW(),
      updated_at = NOW();
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Create trigger to auto-update relationships when conversations are created
DROP TRIGGER IF EXISTS trigger_update_persona_relationship ON conversations;
CREATE TRIGGER trigger_update_persona_relationship
AFTER INSERT ON conversations
FOR EACH ROW
EXECUTE FUNCTION update_persona_relationship();
